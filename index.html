// 全域變數 let currentStaffId = ""; async function handleRoleLogin() { const usernameInput = document.getElementById("login-username"); const roleSelect = document.getElementById("login-role"); const wardSelect = document.getElementById("select-ward"); const staffId = usernameInput.value.trim(); const role = roleSelect.value; const wardId = wardSelect.value; if (!staffId) { alert("請輸入員工編號或帳號！"); usernameInput.focus(); return; } if (!wardId) { alert("請選擇服務單位 / 護理站！"); wardSelect.focus(); return; } const wardName = wardSelect.options[wardSelect.selectedIndex].text; // 1. 同步將員工編號與登入資訊寫入 Supabase if (supabaseClient) { try { const { data, error } = await supabaseClient .from('ward_login_logs') .insert([ { staff_id: staffId, role: role, unit_id: wardId, unit_name: wardName, user_agent: navigator.userAgent } ]); if (error) { console.error("❌ 寫入 Supabase 失敗：", error.message); } else { console.log(`✅ 員工 [${staffId}] 登入紀錄已成功存入 Supabase！`); } } catch (err) { console.warn("連線 Supabase 異常：", err); } } else { console.warn("⚠️ supabaseClient 尚未初始化"); } // 2. 設置系統全域變數 currentStaffId = staffId; currentUserRole = role; currentUnitId = wardId; currentUnitName = wardName; currentUser = currentUserRole === "admin" ? `護理長 (${staffId})` : `護理師 (${staffId})`; // 3. 套用權限樣式 if (currentUserRole === "admin") { document.body.classList.remove("is-user"); document.body.classList.add("is-admin"); } else { document.body.classList.remove("is-admin"); document.body.classList.add("is-user"); } applyRolePermissions(currentUserRole); // 4. 切換頁面進入主系統 document.getElementById("badge-user").textContent = currentUser; document.getElementById("header-unit-switch").value = currentUnitId; document.getElementById("user-badge").style.display = "flex"; document.getElementById("login-section").style.display = "none"; document.getElementById("main-section").style.display = "block"; loadData(); }
